━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🎯 OBJETIVO:
Gerar QR Code PIX de forma segura, sem expor URLs do gateway de pagamento (Ezzpag)
no frontend, e verificar o status do pagamento via polling.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1️⃣ GERAÇÃO DO QR CODE PIX (Backend)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📁 Arquivo: app/api/generate-pix/route.ts

📝 FLUXO:

1. Frontend envia POST para /api/generate-pix:
   {
     "amount": 50.70,
     "customer": { "name": "João", "email": "joao@email.com" },
     "trackingParams": { "gclid": "...", "utm_source": "..." }
   }

2. Backend faz requisição para Ezzpag:
   POST https://api.ezzypag.com.br/v1/transactions
   Headers: Authorization: Bearer {EZZPAG_API_KEY}

3. Ezzpag retorna:
   {
     "transactionId": "121416427",
     "pixCode": "00020126580014br.gov.bcb.pix...",
     "qrCode": "https://api.ezzypag.com.br/qrcode/121416427.png"
   }

4. Backend PROCESSA e ESCONDE informações:
   - Gera QR Code em Base64: QRCode.toDataURL(pixCode)
   - Salva no orderStorage (memória)
   - Envia PENDING para UTMify

5. Backend retorna APENAS:
   {
     "transactionId": "121416427",
     "pixCode": "00020126580014br.gov.bcb.pix...",
     "qrCode": "data:image/png;base64,iVBORw0KGgoAAAA..."
   }

🔒 SEGURANÇA:
✅ URL do Ezzpag NUNCA chega no frontend
✅ API Key fica APENAS no backend
✅ QR Code gerado como Base64

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2️⃣ VERIFICAÇÃO VIA POLLING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📁 Arquivos:
- Frontend: app/checkout/page.tsx
- Backend: app/api/check-transaction-status/route.ts

📝 FLUXO:

1. Frontend inicia polling (10 segundos):
   setInterval(() => {
     fetch('/api/check-transaction-status', {
       body: JSON.stringify({ transactionId: "121416427" })
     })
   }, 10000)

2. Backend consulta Ezzpag:
   GET https://api.ezzypag.com.br/v1/transactions/121416427

3. Se status = "paid":
   - Envia PAID para UTMify
   - Marca como processado
   - Retorna { status: 'paid' }

4. Frontend recebe "paid":
   - Para polling
   - Captura UTMs da URL
   - Redireciona para whitepage com todos os params

🔒 SEGURANÇA:
✅ Frontend NUNCA acessa Ezzpag
✅ Polling passa apenas transactionId
✅ Anti-duplicação via cache

UTMIFY_API_TOKEN=sua_token
